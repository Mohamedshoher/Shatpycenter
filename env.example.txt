# Supabase Configuration
# انسخ هذا الملف باسم .env.local واملأ القيم الحقيقية من لوحة تحكم Supabase

NEXT_PUBLIC_SUPABASE_URL=https://your-project.supabase.co
NEXT_PUBLIC_SUPABASE_ANON_KEY=your_supabase_anon_key_here

# 🔐 سر توقيع جلسات تسجيل الدخول (JWT) — إلزامي، وإلا سيرفض الخادم كل تسجيل دخول
# ولّد قيمة عشوائية طويلة (32 حرفاً على الأقل)، مثال:
#   openssl rand -base64 32

# 🔑 كلمة مرور المدير العام (بدل القيمة الثابتة القديمة في الكود)

# 📞 رقم واتساب المركز (للدعم والمحادثة المباشرة)
# مثال: 201234567890 (رقم مصري بدون صفر)

# 🔗 رابط جروب واتساب الخاص بالمركز (لانضمام أولياء الأمور)
# مثال: https://chat.whatsapp.com/XXXXXXXXXXX
NEXT_PUBLIC_WHATSAPP_GROUP_LINK=your_whatsapp_group_invite_link_here
